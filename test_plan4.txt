4. BUT we also MUST ensure the user cannot log back in automatically via Firebase `onAuthStateChanged`.
Wait, if `onLogout` calls `signOut(auth)`, Firebase will clear the session. So they won't automatically log in.
Wait, what if they close the app and reopen it, and they were banned while they were offline?
If they were offline, `banned_logout` socket event is missed.
Then when they open the app, Firebase logs them in automatically via `onAuthStateChanged`!
We MUST perform a ban check in `app/page.tsx` on `onAuthStateChanged`.
Let's see:
```tsx
  useEffect(() => {
    const checkUserStatus = async (user: any, fallbackUid?: string) => {
      // Get account ID
      const uid = user ? user.uid : fallbackUid;
      const accountId = localStorage.getItem('accountNumber');
      if (uid || accountId) {
        try {
          const { Device } = await import('@capacitor/device');
          const deviceIdInfo = await Device.getId();
          const deviceId = deviceIdInfo.identifier;
          const { apiUrl } = await import('@/src/lib/api');
          const res = await fetch(apiUrl('/api/check-ban'), {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ accountId: accountId || 'N/A', deviceId })
          });
          if (res.ok) {
            const data = await res.json();
            if (data.banned) {
              const type = data.banData.type || 'Violation';
              let unbanTimeStr = 'Never';
              if (data.banData.unbanTime !== -1) {
                unbanTimeStr = new Date(data.banData.unbanTime).toLocaleString();
              }
              localStorage.setItem('recentBanMessage', JSON.stringify(data.banData));
              handleLogout();
              return;
            }
          }
        } catch (e) {}
      }
      setIsLoggedIn(true);
      setLoading(false);
    };

    const unsubscribe = onAuthStateChanged(auth, (user) => {
      if (user) {
        checkUserStatus(user);
      } else {
        // Fallback to LocalStorage Check
        const userUID = localStorage.getItem('userUID')
        const userEmail = localStorage.getItem('userEmail')
        const userPhone = localStorage.getItem('userPhone')

        if (userUID || userEmail || userPhone) {
          checkUserStatus(null, userUID || '');
        } else {
          setIsLoggedIn(false)
          setLoading(false)
        }
      }
    });

    return () => unsubscribe();
  }, [])
```
